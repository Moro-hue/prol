    <script>
        const PASS = "No415963#"; // يمكنك تغيير كلمة السر هنا
        const song = document.getElementById('loveSong');

        function unlock() {
            let val = document.getElementById('passInput').value;
            if(val === PASS) {
                // تشغيل الموسيقى
                song.play().catch(() => console.log("تحتاج تفاعل"));
                document.getElementById('music-btn').style.display = 'flex';
                
                // تفعيل الأجواء
                showPage(2);
                startHearts();
                document.getElementById('bg').style.filter = "blur(0px)";
            } else {
                document.getElementById('errorMsg').style.display = 'block';
            }
        }

        function showPage(n) {
            document.querySelectorAll('.page').forEach(p => p.classList.remove('active'));
            document.getElementById('page' + n).classList.add('active');
        }

        function zoomImg(el) {
            const modal = document.getElementById('imgModal');
            document.getElementById('modalImg').src = el.src;
            modal.style.display = 'flex';
        }

        function toggleMusic() {
            if (song.paused) song.play();
            else song.pause();
        }

        function startHearts() {
            setInterval(() => {
                const heart = document.createElement('div');
                heart.classList.add('heart');
                heart.innerHTML = ['💋','❤️‍🩹','💤','🤭'][Math.floor(Math.random()*4)];
                heart.style.left = Math.random() * 100 + 'vw';
                heart.style.fontSize = (Math.random() * 20 + 15) + 'px';
                heart.style.animationDuration = (Math.random() * 2 + 3) + 's';
                document.body.appendChild(heart);
                setTimeout(() => heart.remove(), 4000);
            }, 300);
        }
    </script>
